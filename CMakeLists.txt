cmake_minimum_required(VERSION 3.22)
project(DuckPocket VERSION 1.0.0)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED JUCE_DIR)
 set(JUCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../JUCE")
endif()
add_subdirectory(${JUCE_DIR} JUCE)
juce_add_plugin(DuckPocket COMPANY_NAME "Rainline Music" COMPANY_WEBSITE "https://github.com/RainlineMusic/phase-pocket" COMPANY_EMAIL "rainlinemusic@gmail.com" BUNDLE_ID "ru.rainlinemusic.duckpocket" AAX_IDENTIFIER "ru.rainlinemusic.duckpocket" PLUGIN_MANUFACTURER_CODE RnLn PLUGIN_CODE DkPk FORMATS VST3 AAX Standalone PRODUCT_NAME "Duck Pocket" DESCRIPTION "Lookahead dynamic sidechain ducker" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Dynamics AAX_CATEGORY Dynamics)
juce_generate_juce_header(DuckPocket)
target_sources(DuckPocket PRIVATE Source/SidechainFilter.h Source/PocketDSP.h Source/PluginProcessor.cpp Source/PluginProcessor.h Source/PluginEditor.cpp Source/PluginEditor.h)
target_compile_definitions(DuckPocket PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 $<$<BOOL:${WIN32}>:NOMINMAX=1> $<$<BOOL:${WIN32}>:WIN32_LEAN_AND_MEAN=1>)
if(MSVC)
 target_compile_options(DuckPocket PRIVATE /utf-8)
else()
 target_compile_options(DuckPocket PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
endif()
target_link_libraries(DuckPocket PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_lto_flags juce::juce_recommended_warning_flags)
add_executable(PocketV10Test Tests/v10_test.cpp)
add_executable(PocketShortDurationTest Tests/short_duration_test.cpp)
if(MSVC)
 target_compile_options(PocketV10Test PRIVATE /utf-8)
 target_compile_options(PocketShortDurationTest PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME PocketV10 COMMAND PocketV10Test)
add_test(NAME PocketShortDuration COMMAND PocketShortDurationTest)
